export type DemoOrder = { code: string; name: string; email: string; pass: PassId; day: string; qty: number; total: number; createdAt: string };
export const CART_KEY = "umbral:selection:v1";
export const ORDER_KEY = "umbral:order:v1";
export function isOrder(value: unknown): value is DemoOrder {
  if (!value || typeof value !== "object") return false;
  const v = value as Partial<DemoOrder>;
  const pass = PASSES.find((p) => p.id === v.pass);
  return !!pass && typeof v.code === "string" && /^UMB-27-[A-Z0-9]{6}$/.test(v.code) && typeof v.name === "string" &&
    typeof v.email === "string" && Number.isInteger(v.qty) && v.qty! >= 1 && v.qty! <= 6 &&
    DAYS.some((day) => day.value === v.day) && v.total === pass.price * v.qty! && typeof v.createdAt === "string";
}
