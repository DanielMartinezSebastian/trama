"use client";
import { useState } from "react";
export default function ForumVisit() {
  const [done, setDone] = useState(false);
  function submit(e: React.FormEvent<HTMLFormElement>) { e.preventDefault(); const fd = new FormData(e.currentTarget); const records = JSON.parse(localStorage.getItem("rizo-forum-visits") ?? "[]") as unknown[]; localStorage.setItem("rizo-forum-visits", JSON.stringify([...records, {name:fd.get("name"), email:fd.get("email"), day:fd.get("day"), time:fd.get("time"), at:new Date().toISOString()}])); setDone(true); }
  return <form className="forum-visit" onSubmit={submit}>{done ? <div className="visit-success" role="status"><span>✓</span><div><strong>Tu visita de prueba está en este navegador.</strong><p>FORO / 10 y su agenda son una simulación. No se ha enviado una reserva real.</p><button type="button" onClick={() => setDone(false)}>Crear otra visita</button></div></div> : <><label>Nombre<input name="name" required minLength={2} autoComplete="name" placeholder="Alex García"/></label><label>Correo de trabajo<input name="email" type="email" required autoComplete="email" placeholder="alex@empresa.com"/></label><div className="visit-row"><label>Día<select name="day"><option>Jueves 12 de noviembre</option><option>Viernes 13 de noviembre</option></select></label><label>Franja<select name="time"><option>10:00—13:00</option><option>13:00—16:00</option><option>16:00—19:00</option></select></label></div><button type="submit" className="button button--acid">Reservar visita de prueba ↗</button><p className="form-note">Sin pagos ni confirmación por correo. Solo una visita ficticia en tu navegador.</p></>}</form>;
}
